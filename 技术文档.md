## 一、魔法方法（Magic Methods / Dunder Methods）

### 1.1 什么是魔法方法？

魔法方法是 Python 中以双下划线开头和结尾的特殊方法，它们让你的类能够：
- 自定义对象的创建和销毁
- 重载运算符
- 实现容器协议
- 支持上下文管理器
- 支持迭代器协议

### 1.2 常用魔法方法分类

#### 对象生命周期
```python
class MyClass:
    def __new__(cls, *args, **kwargs):
        """创建实例（在 __init__ 之前调用）"""
        print("__new__ 被调用")
        instance = super().__new__(cls)
        return instance

    def __init__(self, value):
        """初始化实例"""
        print("__init__ 被调用")
        self.value = value

    def __del__(self):
        """析构函数（对象被垃圾回收时调用）"""
        print(f"__del__ 被调用: {self.value}")
```

**面试考点：**
- `__new__` 是静态方法，`__init__` 是实例方法
- `__new__` 返回实例，`__init__` 不返回
- 单例模式通常在 `__new__` 中实现

#### 字符串表示
```python
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __str__(self):
        """str() 和 print() 时调用，面向用户"""
        return f"Point({self.x}, {self.y})"

    def __repr__(self):
        """repr() 和交互式环境显示，面向开发者"""
        return f"Point(x={self.x}, y={self.y})"
```

**面试考点：**
- `__str__` 面向用户，`__repr__` 面向开发者
- 如果只实现一个，优先实现 `__repr__`
- `__str__` 默认回退到 `__repr__`

#### 比较运算符
```python
class Number:
    def __init__(self, value):
        self.value = value

    def __eq__(self, other):    # ==
        return self.value == other.value

    def __ne__(self, other):    # !=
        return self.value != other.value

    def __lt__(self, other):    # <
        return self.value < other.value

    def __le__(self, other):    # <=
        return self.value <= other.value

    def __gt__(self, other):    # >
        return self.value > other.value

    def __ge__(self, other):    # >=
        return self.value >= other.value

    def __hash__(self):         # hash()
        return hash(self.value)
```

**面试考点：**
- 实现 `__eq__` 后，`__hash__` 会被设为 `None`（除非显式实现）
- `functools.total_ordering` 可以只实现 `__eq__` 和一个比较方法

#### 算术运算符
```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __add__(self, other):       # +
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other):       # -
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, scalar):      # *
        return Vector(self.x * scalar, self.y * scalar)

    def __rmul__(self, scalar):     # 右乘：scalar * vector
        return self.__mul__(scalar)

    def __truediv__(self, scalar):  # /
        return Vector(self.x / scalar, self.y / scalar)

    def __neg__(self):              # -vector
        return Vector(-self.x, -self.y)

    def __abs__(self):              # abs(vector)
        return (self.x ** 2 + self.y ** 2) ** 0.5
```

#### 容器协议
```python
class CustomList:
    def __init__(self, data):
        self._data = list(data)

    def __len__(self):              # len()
        return len(self._data)

    def __getitem__(self, index):   # obj[index]
        return self._data[index]

    def __setitem__(self, index, value):  # obj[index] = value
        self._data[index] = value

    def __delitem__(self, index):   # del obj[index]
        del self._data[index]

    def __contains__(self, item):   # in
        return item in self._data

    def __iter__(self):             # for 循环
        return iter(self._data)

    def __reversed__(self):         # reversed()
        return reversed(self._data)
```

#### 上下文管理器
```python
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None

    def __enter__(self):
        """进入 with 语句时调用"""
        print(f"打开文件: {self.filename}")
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出 with 语句时调用"""
        print(f"关闭文件: {self.filename}")
        if self.file:
            self.file.close()
        # 返回 True 会抑制异常，返回 False 会传播异常
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello")
```

**面试考点：**
- `__enter__` 返回值赋给 `as` 后的变量
- `__exit__` 参数：异常类型、异常值、 traceback
- 返回 `True` 抑制异常，`False` 传播异常

#### 属性访问控制
```python
class DynamicAttr:
    def __init__(self):
        self._data = {}

    def __getattr__(self, name):
        """访问不存在的属性时调用"""
        print(f"__getattr__: {name}")
        return self._data.get(name)

    def __setattr__(self, name, value):
        """设置属性时调用"""
        print(f"__setattr__: {name} = {value}")
        if name.startswith('_'):
            super().__setattr__(name, value)
        else:
            self._data[name] = value

    def __delattr__(self, name):
        """删除属性时调用"""
        print(f"__delattr__: {name}")
        if name in self._data:
            del self._data[name]

    def __getattribute__(self, name):
        """访问任何属性时都会调用（注意：容易死循环）"""
        if name == '_data':
            return super().__getattribute__(name)
        print(f"__getattribute__: {name}")
        return super().__getattribute__(name)
```

**面试考点：**
- `__getattr__` 只在属性不存在时调用
- `__getattribute__` 每次访问都调用，容易死循环
- 属性查找顺序：`__getattribute__` → 数据描述符 → 实例字典 → 类字典 → `__getattr__`

#### 描述符协议
```python
class Property:
    """描述符 — 实现 __get__, __set__, __delete__"""
    def __init__(self, fget=None, fset=None, fdel=None):
        self.fget = fget
        self.fset = fset
        self.fdel = fdel

    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return self.fget(obj)

    def __set__(self, obj, value):
        if self.fset is None:
            raise AttributeError("can't set attribute")
        self.fset(obj, value)

    def __delete__(self, obj):
        if self.fdel is None:
            raise AttributeError("can't delete attribute")
        self.fdel(obj)

    def getter(self, fget):
        return Property(fget, self.fset, self.fdel)

    def setter(self, fset):
        return Property(self.fget, fset, self.fdel)

    def deleter(self, fdel):
        return Property(self.fget, self.fset, fdel)
```

## 二、装饰器（Decorators）

### 2.1 基础装饰器

```python
def timer(func):
    """计时装饰器"""
    import time
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} 耗时: {end - start:.4f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(1)
```

### 2.2 带参数的装饰器

```python
def retry(max_attempts=3, delay=1):
    """重试装饰器"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt < max_attempts - 1:
                        time.sleep(delay)
                        continue
                    raise
        return wrapper
    return decorator

@retry(max_attempts=3, delay=0.5)
def unstable_function():
    import random
    if random.random() < 0.5:
        raise ValueError("随机失败")
    return "成功"
```

### 2.3 类装饰器

```python
class Singleton:
    """单例装饰器"""
    def __init__(self, cls):
        self.cls = cls
        self.instance = None

    def __call__(self, *args, **kwargs):
        if self.instance is None:
            self.instance = self.cls(*args, **kwargs)
        return self.instance

@Singleton
class Database:
```

### 2.4 functools.wraps 的作用

```python

def my_decorator(func):
```

### 3.5 Task 和 Future

```python
async def fetch_data(url):
    """模拟异步请求"""
```

### 3.6 异步队列

```python
async def producer(queue, n):
    for i in range(n):
        await asyncio.sleep(0.1)
        await queue.put(i)
        print(f"生产: {i}")
    await queue.put(None)  # 结束信号

async def consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"消费: {item}")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue(maxsize=5)
    await asyncio.gather(
        producer(queue, 10),
        consumer(queue),
    )
```

```python
class SingletonMeta(type):
    """单例元类"""
    _instances = {}

    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class Database(metaclass=SingletonMeta):
    def __init__(self):
        print("初始化数据库")
```

## 五、GIL（全局解释器锁）

### 5.1 什么是 GIL？

```python
import threading

counter = 0

def increment():
    global counter
    for _ in range(1000000):
        counter += 1  # 非线程安全！

threads = [threading.Thread(target=increment) for _ in range(4)]
for t in threads:
    t.start()
for t in threads:
    t.join()

print(f"预期: 4000000, 实际: {counter}")  # 通常小于 4000000
```

### 5.2 绕过 GIL

```python
# 方法 1: 多进程
from multiprocessing import Pool

def cpu_bound_task(n):
    return sum(i * i for i in range(n))

with Pool(4) as p:
    results = p.map(cpu_bound_task, [10**6] * 4)

# 方法 2: 使用 C 扩展（NumPy 等）
import numpy as np
arr = np.random.rand(1000000)
result = np.sum(arr)  # 释放 GIL

# 方法 3: IO 密集型使用 asyncio
import asyncio
```

### 5.3 GIL 面试要点

- GIL 只存在于 CPython（官方实现）
- GIL 保护 Python 对象的内存安全
- IO 密集型任务：用 asyncio 或 threading
- CPU 密集型任务：用 multiprocessing 或 C 扩展
- Python 3.12+ 改进了 GIL 的粒度

### 7.3 观察者模式

```python
class EventEmitter:
    def __init__(self):
        self._listeners = {}

    def on(self, event, callback):
        if event not in self._listeners:
            self._listeners[event] = []
        self._listeners[event].append(callback)

    def emit(self, event, *args, **kwargs):
        for callback in self._listeners.get(event, []):
            callback(*args, **kwargs)

    def off(self, event, callback):
        if event in self._listeners:
            self._listeners[event].remove(callback)

# 使用
emitter = EventEmitter()
emitter.on("data", lambda x: print(f"收到: {x}"))
emitter.emit("data", 42)
```

### 7.4 策略模式

```python
from abc import ABC, abstractmethod

class SortStrategy(ABC):
    @abstractmethod
    def sort(self, data):
        pass

class BubbleSort(SortStrategy):
    def sort(self, data):
        # 冒泡排序实现
        return sorted(data)  # 简化

class QuickSort(SortStrategy):
    def sort(self, data):
        # 快速排序实现
        return sorted(data)  # 简化

class Sorter:
    def __init__(self, strategy: SortStrategy):
        self.strategy = strategy

    def sort(self, data):
        return self.strategy.sort(data)

# 使用
sorter = Sorter(BubbleSort())
print(sorter.sort([3, 1, 4, 1, 5]))
```

### 9.1 实现 LRU Cache

```python
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.cache = OrderedDict()
        self.capacity = capacity

    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)
```

### 9.2 实现线程安全的队列

```python
import threading
from collections import deque

class ThreadSafeQueue:
    def __init__(self, maxsize=0):
        self.queue = deque()
        self.maxsize = maxsize
        self.not_empty = threading.Condition()
        self.not_full = threading.Condition()

    def put(self, item):
        with self.not_full:
            if self.maxsize > 0:
                while len(self.queue) >= self.maxsize:
                    self.not_full.wait()
            self.queue.append(item)
            self.not_empty.notify()

    def get(self):
        with self.not_empty:
            while len(self.queue) == 0:
                self.not_empty.wait()
            item = self.queue.popleft()
            self.not_full.notify()
            return item
```

### 9.3 实现上下文管理器装饰器

```python
from contextlib import contextmanager
import time

@contextmanager
def timer(label=""):
    start = time.perf_counter()
    try:
        yield
    finally:
        elapsed = time.perf_counter() - start
        print(f"{label} 耗时: {elapsed:.4f}s")

# 使用
with timer("排序"):
    sorted(range(1000000))
```

### 9.4 实现异步任务调度器

```python
import asyncio
from typing import Any, Callable, Coroutine

class AsyncTaskScheduler:
    def __init__(self, max_concurrent=5):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.tasks = []

    async def run(self, coro: Coroutine) -> Any:
        async with self.semaphore:
            return await coro

    async def gather(self, *coros):
        tasks = [self.run(coro) for coro in coros]
        return await asyncio.gather(*tasks)

# 使用
async def main():
    scheduler = AsyncTaskScheduler(max_concurrent=3)

    async def task(i):
        await asyncio.sleep(1)
        return i * 2

    results = await scheduler.gather(*[task(i) for i in range(10)])
    print(results)
```

### 10.1 使用生成器节省内存

```python
def get_data_bad():
```
